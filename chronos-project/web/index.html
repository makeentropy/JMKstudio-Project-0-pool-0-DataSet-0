        📱 Termux 使用指南

        # 1. 安装依赖
pkg install python python-numpy
pip install prompt-toolkit

# 2. 启动 CLI
cd chronos-project/cli
python chronos_cli.py

# 3. 终端 UI 模式 (curses)
python chronos_cli.py tui

# 4. 导出数据 (供 Web 导入)
python chronos_cli.py export > data.json

# 5. 导入 Web 导出的数据
python chronos_cli.py import web_data.json
